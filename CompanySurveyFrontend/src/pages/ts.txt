/* This page will:

Fetch the published survey
Fetch all its questions
Render them dynamically
Let employees answer
Submit responses */
import { useEffect, useState } from "react";
import { useParams } from "react-router-dom";
import * as API from "../services/api";

/* ===========================================
   Survey Helper Functions
=========================================== */

function sortQuestions(questions) {

    return [...questions].sort((firstQuestion, secondQuestion) =>

        (firstQuestion.display_order || 0) -
        (secondQuestion.display_order || 0)

    );

}


// ===========================================
// Get Question Options
// ===========================================
function getQuestionOptions(question) {

    const rawOptions =
        question.options ||
        question.choices ||
        question.answer_options ||
        [];

    if (Array.isArray(rawOptions) && rawOptions.length > 0) {

        return rawOptions.map((option) => {

            if (typeof option === "string" || typeof option === "number") {

                return {

                    label: String(option),

                    value: String(option),

                };

            }

            return {

                label:
                    option.label ||
                    option.text ||
                    option.name ||
                    option.value ||
                    "Option",

                value: String(
                    option.value ??
                    option.id ??
                    option.label ??
                    option.text ??
                    option.name ??
                    ""
                ),

            };

        });

    }

    // Default options
    return [

        { label: "Option 1", value: "Option 1" },

        { label: "Option 2", value: "Option 2" },

        { label: "Option 3", value: "Option 3" },

    ];

}


// ===========================================
// Rating Scale
// ===========================================
function getRatingScale(question) {

    const scaleValue = Number(

        question.rating_scale ||
        question.scale ||
        5

    );

    if (Number.isNaN(scaleValue)) {

        return 5;

    }

    return Math.min(Math.max(scaleValue, 1), 10);

}


// ===========================================
// Take Survey Component
// ===========================================
function TakeSurvey() {

    const { id } = useParams();

    const [survey, setSurvey] = useState(null);

    const [questions, setQuestions] = useState([]);

    const [answers, setAnswers] = useState({});

    const [loading, setLoading] = useState(true);

    const [error, setError] = useState("");


// ===========================================
// Handle Answer Change
// ===========================================
    function handleAnswerChange(questionId, value) {

        setAnswers((previousAnswers) => ({

            ...previousAnswers,

            [questionId]: value,

        }));

    }

// ===========================================
// Checkbox Handler
// ===========================================
function handleCheckboxChange(questionId, optionValue, checked) {

    setAnswers((prev) => {

        const currentValues = Array.isArray(prev[questionId])
            ? prev[questionId]
            : [];

        return {

            ...prev,

            [questionId]: checked
                ? [...currentValues, optionValue]
                : currentValues.filter((value) => value !== optionValue),

        };

    });

}


// ===========================================
// Rating Handler
// ===========================================
function handleRatingChange(questionId, rating) {

    handleAnswerChange(questionId, String(rating));

}


// ===========================================
// Submit Survey
// ===========================================
async function handleSubmit(event) {

    event.preventDefault();

    // Validate required questions
    const missingRequiredQuestion = questions.find((question) => {

        if (!question.is_required) {
            return false;
        }

        const currentAnswer = answers[question.id];

        if (Array.isArray(currentAnswer)) {
            return currentAnswer.length === 0;
        }

        return (
            currentAnswer === undefined ||
            currentAnswer === null ||
            String(currentAnswer).trim() === ""
        );

    });

    if (missingRequiredQuestion) {

        setError(
            `Please answer: ${missingRequiredQuestion.question_text}`
        );

        return;

    }

    setError("");

    try {

        // Convert answers into backend payload
        const payload = Object.entries(answers).map(
                ([questionId, answer]) => ({
                question_id: Number(questionId),
                answer: Array.isArray(answer) ? answer.join(", ") : String(answer),
            })
        );

        // TakeSurvey.jsx line 208
        const response = await API.submitSurvey(id, payload);

        const data = await response.json();

        if (!response.ok) {

            const errorMessage = Array.isArray(data.detail)
        ? data.detail.map((err) => `${err.loc.join('.')}: ${err.msg}`).join('\n')
        : data.detail || "Submission failed.";

            setError(errorMessage);

            return;

        }

        alert("Survey submitted successfully!");

    }

    catch (error) {

        console.error("Survey Submission Error:", error);

        alert("Something went wrong while submitting the survey.");

    }

}


// ===========================================
// Load Survey
// ===========================================
// ===========================================
// Load Survey
// ===========================================
useEffect(() => {
    async function fetchSurvey() {
        try {
            setLoading(true);
            setError("");

            // API.getSurveyQuestions already returns JSON from api.js
            const data = await API.getSurveyQuestions(id);

            console.log("API Response Data:", data);

            // Handle both object wrapper and direct array formats
            if (Array.isArray(data)) {
                setSurvey({ title: "Survey" });
                setQuestions(sortQuestions(data));
            } else if (data && Array.isArray(data.questions)) {
                setSurvey(data);
                setQuestions(sortQuestions(data.questions));
            } else {
                setSurvey(data || null);
                setQuestions([]);
            }
        } catch (fetchError) {
            console.error("Survey Load Error:", fetchError);
            setError("Failed to load survey questions.");
        } finally {
            setLoading(false);
        }
    }

    fetchSurvey();
}, [id]);

// ===========================================
// Loading Screen
// ===========================================
if (loading) {

    return <h2>Loading Survey...</h2>;

}
if (!loading && questions.length === 0) {

    return (

        <div className="take-survey">

            <h2>No Questions Found</h2>

            <p>
                This survey currently has no questions or the API returned an empty list.
            </p>

        </div>

    );

}
return (
    <div className="take-survey">

        <h1>{survey?.title || "Survey"}</h1>
        <p>{survey?.description}</p>

        {error && (
            <p style={{ color: "#b00020" }}>
                {error}
            </p>
        )}

        {/* Temporary debug */}
        <p>Total Questions: {questions.length}</p>

        <hr />

        <form onSubmit={handleSubmit}>

            {questions.map((question) => {

                const questionType = question.question_type || question.type || "text";
                const questionOptions = getQuestionOptions(question);
                const ratingScale = getRatingScale(question);
                const currentAnswer = answers[question.id];

                return (

                    <div key={question.id} className="question-card">

                        <h3>
                            {question.question_text}
                            {question.is_required && <span style={{ color: "red" }}> *</span>}
                        </h3>

                        {/* TEXT */}
                        {questionType === "text" && (
                            <input
                                type="text"
                                placeholder="Enter your answer..."
                                value={currentAnswer || ""}
                                onChange={(e) =>
                                    handleAnswerChange(question.id, e.target.value)
                                }
                            />
                        )}

                        {/* TEXTAREA */}
                        {questionType === "textarea" && (
                            <textarea
                                rows={4}
                                placeholder="Enter your answer..."
                                value={currentAnswer || ""}
                                onChange={(e) =>
                                    handleAnswerChange(question.id, e.target.value)
                                }
                            />
                        )}

                        {/* RADIO */}
                        {questionType === "radio" && (
                            <>
                                {questionOptions.map((option) => (
                                    <label
                                        key={option.value}
                                        style={{
                                            display: "block",
                                            marginBottom: "8px",
                                        }}
                                    >
                                        <input
                                            type="radio"
                                            name={`question-${question.id}`}
                                            value={option.value}
                                            checked={String(currentAnswer) === String(option.value)}
                                            onChange={(e) =>
                                                handleAnswerChange(
                                                    question.id,
                                                    e.target.value
                                                )
                                            }
                                        />
                                        {" "}
                                        {option.label}
                                    </label>
                                ))}
                            </>
                        )}

                        {/* CHECKBOX */}
                        {questionType === "checkbox" && (
                            <>
                                {questionOptions.map((option) => {

                                    const selected =
                                        Array.isArray(currentAnswer)
                                            ? currentAnswer
                                            : [];

                                    return (
                                        <label
                                            key={option.value}
                                            style={{
                                                display: "block",
                                                marginBottom: "8px",
                                            }}
                                        >
                                            <input
                                                type="checkbox"
                                                checked={selected.includes(option.value)}
                                                onChange={(e) =>
                                                    handleCheckboxChange(
                                                        question.id,
                                                        option.value,
                                                        e.target.checked
                                                    )
                                                }
                                            />
                                            {" "}
                                            {option.label}
                                        </label>
                                    );
                                })}
                            </>
                        )}

                        {/* DROPDOWN */}
                        {questionType === "dropdown" && (
                            <select
                                value={currentAnswer || ""}
                                onChange={(e) =>
                                    handleAnswerChange(question.id, e.target.value)
                                }
                            >
                                <option value="">
                                    Select an option
                                </option>

                                {questionOptions.map((option) => (
                                    <option
                                        key={option.value}
                                        value={option.value}
                                    >
                                        {option.label}
                                    </option>
                                ))}
                            </select>
                        )}

                        {/* RATING */}
                        {questionType === "rating" && (
                            <div
                                style={{
                                    display: "flex",
                                    gap: "8px",
                                    flexWrap: "wrap",
                                }}
                            >
                                {Array.from(
                                    { length: ratingScale },
                                    (_, index) => {

                                        const value = index + 1;

                                        return (
                                            <button
                                                key={value}
                                                type="button"
                                                onClick={() =>
                                                    handleRatingChange(
                                                        question.id,
                                                        value
                                                    )
                                                }
                                            >
                                                {value}
                                            </button>
                                        );
                                    }
                                )}
                            </div>
                        )}

                    </div>

                );
            })}

            <div style={{ marginTop: "20px" }}>
                <button type="submit">
                    Submit Survey
                </button>
            </div>

        </form>

    </div>
);
}

export default TakeSurvey;
